# CUDA platform for ConstantVoltage plugin

# Include CUDA related files
INCLUDE_DIRECTORIES(${CUDAToolkit_LIBRARY_DIR})
INCLUDE_DIRECTORIES(${CMAKE_CURRENT_SOURCE_DIR}/include)
INCLUDE_DIRECTORIES(${CMAKE_CURRENT_SOURCE_DIR}/../common/include)

# Set variables for kernel encoding
SET(CONSTANTV_CUDA_KERNEL_SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src)
SET(CONSTANTV_CUDA_KERNEL_SOURCE_CLASS ConstantVoltageKernelSources)
SET(CONSTANTV_CUDA_KERNELS_CPP ${CMAKE_CURRENT_BINARY_DIR}/src/${CONSTANTV_CUDA_KERNEL_SOURCE_CLASS}.cpp)
SET(CONSTANTV_CUDA_KERNELS_H ${CMAKE_CURRENT_BINARY_DIR}/src/${CONSTANTV_CUDA_KERNEL_SOURCE_CLASS}.h)

# Encode kernel sources
FILE(GLOB CONSTANTV_CUDA_KERNELS ${CONSTANTV_CUDA_KERNEL_SOURCE_DIR}/kernels/*.cu)
ADD_CUSTOM_COMMAND(OUTPUT ${CONSTANTV_CUDA_KERNELS_CPP} ${CONSTANTV_CUDA_KERNELS_H}
    COMMAND ${CMAKE_COMMAND}
    ARGS -D KERNEL_SOURCE_DIR=${CONSTANTV_CUDA_KERNEL_SOURCE_DIR} -D KERNELS_CPP=${CONSTANTV_CUDA_KERNELS_CPP} -D KERNELS_H=${CONSTANTV_CUDA_KERNELS_H} -D KERNEL_SOURCE_CLASS=${CONSTANTV_CUDA_KERNEL_SOURCE_CLASS} -D KERNEL_FILE_EXTENSION=cu -P ${CMAKE_SOURCE_DIR}/cmake_modules/EncodeKernelFiles.cmake
    DEPENDS ${CONSTANTV_CUDA_KERNELS}
)
SET_SOURCE_FILES_PROPERTIES(${CONSTANTV_CUDA_KERNELS_CPP} ${CONSTANTV_CUDA_KERNELS_H} PROPERTIES GENERATED TRUE)

INCLUDE_DIRECTORIES(BEFORE ${CMAKE_CURRENT_BINARY_DIR}/src)

SET(CUDA_SOURCE_FILES
    ${CMAKE_CURRENT_SOURCE_DIR}/src/CudaConstantVoltageKernels.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/CudaConstantVoltageKernelFactory.cpp
    ${CONSTANTV_CUDA_KERNELS_CPP}
    ${CONSTANTV_CUDA_KERNELS_H}
)

# Create CUDA library for ConstantVoltage
SET(OPENMM_CONSTANTV_CUDA_LIBRARY_NAME OpenMMConstantVoltageCUDA)
ADD_LIBRARY(${OPENMM_CONSTANTV_CUDA_LIBRARY_NAME} SHARED ${CUDA_SOURCE_FILES})

TARGET_LINK_LIBRARIES(${OPENMM_CONSTANTV_CUDA_LIBRARY_NAME} ${SHARED_CONSTANTV_TARGET} OpenMMCUDA CUDA::cuda_driver ${NVRTC_LIB})
SET_TARGET_PROPERTIES(${OPENMM_CONSTANTV_CUDA_LIBRARY_NAME} PROPERTIES 
    COMPILE_FLAGS "${EXTRA_COMPILE_FLAGS} -DOPENMM_CONSTANTV_BUILDING_SHARED_LIBRARY"
    LINK_FLAGS "${EXTRA_LINK_FLAGS}")

INSTALL_TARGETS(/lib/plugins RUNTIME_DIRECTORY /lib/plugins ${OPENMM_CONSTANTV_CUDA_LIBRARY_NAME})
