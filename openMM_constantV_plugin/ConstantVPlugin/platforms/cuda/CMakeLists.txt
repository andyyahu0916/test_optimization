SET(OPENMM_CONSTANTV_CUDA_LIBRARY_NAME ConstantVPluginCUDA)
SET(SHARED_TARGET ${OPENMM_CONSTANTV_CUDA_LIBRARY_NAME})

# Find CUDA using modern CMake approach
FIND_PACKAGE(CUDAToolkit QUIET)
IF(NOT CUDAToolkit_FOUND)
    MESSAGE(STATUS "CUDA not found - CUDA platform will not be built")
    RETURN()
ENDIF()

# Enable CUDA language
ENABLE_LANGUAGE(CUDA)
SET(CMAKE_CUDA_STANDARD 14)
SET(CMAKE_CUDA_STANDARD_REQUIRED ON)

SET(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -DLEPTON_USE_STATIC_LIBRARIES")

# API include dirs
SET(API_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/include")

# Locate header files
SET(API_INCLUDE_FILES)
FOREACH(dir ${API_INCLUDE_DIRS})
    FILE(GLOB fullpaths ${dir}/*.h)
    SET(API_INCLUDE_FILES ${API_INCLUDE_FILES} ${fullpaths})
ENDFOREACH(dir)

# Collect source files
FILE(GLOB SOURCE_FILES ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
FILE(GLOB SOURCE_FILES_CUDA ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cu)

INCLUDE_DIRECTORIES(BEFORE ${CMAKE_CURRENT_SOURCE_DIR}/include)
INCLUDE_DIRECTORIES(BEFORE ${CMAKE_SOURCE_DIR}/openmmapi/include)
INCLUDE_DIRECTORIES(BEFORE ${CMAKE_SOURCE_DIR}/openmmapi/include/internal)
INCLUDE_DIRECTORIES(BEFORE ${OPENMM_DIR}/include)

# CUDA specific settings
INCLUDE_DIRECTORIES(${CUDAToolkit_INCLUDE_DIRS})

# Set CUDA compilation flags
SET(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} --use_fast_math --allow-unsupported-compiler")

# Add compute capabilities (only modern GPUs for simplicity)
SET(CMAKE_CUDA_ARCHITECTURES 70 75 80 86)

# Create library with both .cpp and .cu files
ADD_LIBRARY(${SHARED_TARGET} SHARED ${SOURCE_FILES} ${SOURCE_FILES_CUDA} ${API_INCLUDE_FILES})

# Set CUDA properties
SET_TARGET_PROPERTIES(${SHARED_TARGET} PROPERTIES
    CUDA_SEPARABLE_COMPILATION ON
    CUDA_RESOLVE_DEVICE_SYMBOLS ON
)

# Link libraries using modern imported targets
TARGET_LINK_LIBRARIES(${SHARED_TARGET} CUDA::cudart CUDA::cublas)
TARGET_LINK_LIBRARIES(${SHARED_TARGET} OpenMM)
TARGET_LINK_LIBRARIES(${SHARED_TARGET} OpenMMCUDA)
TARGET_LINK_LIBRARIES(${SHARED_TARGET} ${SHARED_CONSTANTV_TARGET})

SET_TARGET_PROPERTIES(${SHARED_TARGET} PROPERTIES
    COMPILE_FLAGS "-DOPENMM_BUILDING_SHARED_LIBRARY ${EXTRA_COMPILE_FLAGS}"
    LINK_FLAGS "${EXTRA_COMPILE_FLAGS}")

INSTALL(TARGETS ${SHARED_TARGET} DESTINATION ${CMAKE_INSTALL_PREFIX}/lib/plugins)
